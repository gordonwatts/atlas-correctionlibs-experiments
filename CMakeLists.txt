cmake_minimum_required(VERSION 3.20)
project(AtlasCorrectionLibDemo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_library(LWTNN_LIBRARY lwtnn REQUIRED)
find_package(ROOT REQUIRED COMPONENTS Core Physics MathCore)
find_package(correctionlib CONFIG REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_atlas_corrections src/nominal.cxx)
target_link_libraries(_atlas_corrections PRIVATE ${LWTNN_LIBRARY} Eigen3::Eigen correctionlib ROOT::Core ROOT::Physics ROOT::MathCore)
install(TARGETS _atlas_corrections LIBRARY DESTINATION python)
