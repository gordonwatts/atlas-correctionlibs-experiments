cmake_minimum_required(VERSION 3.20)
project(DeclarativeCorrectionLib LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_library(LWTNN_LIBRARY lwtnn REQUIRED)
find_package(ROOT REQUIRED COMPONENTS Core Physics MathCore)
find_package(pybind11 CONFIG REQUIRED)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CORRECTIONLIB_PYTHON OFF CACHE BOOL "" FORCE)
set(SKBUILD_PROJECT_VERSION "2.7.0")
add_subdirectory(../external/correctionlib-d correctionlib)
pybind11_add_module(_declarative src/bindings.cxx)
target_link_libraries(_declarative PRIVATE correctionlib ${LWTNN_LIBRARY} Eigen3::Eigen ROOT::Core ROOT::Physics ROOT::MathCore)
target_compile_options(_declarative PRIVATE -ffp-contract=off)
